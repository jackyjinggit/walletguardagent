## 二、现成模板（供董鹏参考，不强制）

基于 `web3-30min-onboard` 中已有的 AgentGuard 合约结构：

```solidity
contract AgentGuard {
    address public owner;
    address public executor;
    address public merchant;
    uint256 public price;

    struct Order { uint256 amount; bool paid; bool revoked; }
    mapping(bytes32 => Order) public orders;
    mapping(address => uint256) public spent;
    uint256 public orderCap = 0.025 * 1e18;
    uint256 public totalCap = 0.030 * 1e18;

    modifier onlyOwner() { require(msg.sender == owner); _; }
    modifier onlyExecutor() { require(msg.sender == executor); _; }

    function deposit() public payable onlyOwner { }
    function configure(address _executor, address _merchant, uint256 _price) public onlyOwner { }
    function spend(bytes32 orderId) public onlyExecutor {
        require(!orders[orderId].paid, "already paid");
        require(!orders[orderId].revoked, "revoked");
        require(price <= orderCap, "over single cap");
        require(spent[msg.sender] + price <= totalCap, "over total cap");
        orders[orderId] = Order(price, true, false);
        spent[msg.sender] += price;
        payable(merchant).transfer(price);
        emit OrderPaid(orderId, price);
    }
    function revoke(bytes32 orderId) public onlyOwner { }
    function withdraw() public onlyOwner { }
}
```

**董鹏需新增的部分**（决策可验证）：

```solidity
event DecisionRecorded(bytes32 indexed orderId, bytes32 decisionHash, uint256 timestamp);

function recordDecision(bytes32 orderId, bytes32 decisionHash) public onlyOwner {
    emit DecisionRecorded(orderId, decisionHash, block.timestamp);
}
```
